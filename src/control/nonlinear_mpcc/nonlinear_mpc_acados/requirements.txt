# ====================================================================
# ACADOS — REQUIRED (manual install, not pip)
# ====================================================================
# 1) Clone:
#    cd ~ && git clone https://github.com/acados/acados.git
#    cd acados && git submodule update --recursive --init
#
# 2) Build (CMake):
#    mkdir -p build && cd build
#    cmake -DACADOS_WITH_QPOASES=ON \
#          -DACADOS_WITH_OSQP=ON \
#          -DACADOS_INSTALL_DIR="${HOME}/acados" \
#          ..
#    make install -j4
#
# 3) Python interface:
#    pip install -e ~/acados/interfaces/acados_template \
#                --user --break-system-packages
#
# 4) Environment (~/.bashrc):
#    export ACADOS_SOURCE_DIR=~/acados
#    export LD_LIBRARY_PATH=~/acados/lib:$LD_LIBRARY_PATH
#
# 5) Verify:
#    python3 -c "from acados_template import AcadosOcp; print('OK')"
#
# Documentation: https://docs.acados.org/installation/
# ====================================================================
